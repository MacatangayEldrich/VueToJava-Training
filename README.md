# VueToJava-Training

## Running the Application
#### Start the Backend
```
cd backend
mvn spring-boot:run
```

#### Start the Frontend
```
cd frontend
npm install
npm run dev
```

```
SELECT PO_ID AS REQUEST_ID, 'Overtime Request' AS TYPE, FILLING_DATE, STATUS AS STATUS FROM PLANNED_OVERTIME
UNION ALL
SELECT AO_ID AS REQUEST_ID, 'Actual Overtime' AS TYPE, FILLING_DATE,  STATUS AS STATUS FROM ACTUAL_OVERTIME
UNION ALL
SELECT UO_ID AS REQUEST_ID, 'CTO Request' AS TYPE, FILLING_DATE, UO_STATUS AS STATUS FROM USE_OVERTIME
ORDER BY FILLING_DATE;

SELECT p.PO_ID AS REQUEST_ID, 'Overtime Request' AS TYPE, e.EMPLOYEE_NAME, p.FILLING_DATE, p.STATUS AS STATUS FROM PLANNED_OVERTIME p
INNER JOIN EMPLOYEE e
  ON p.EMPLOYEE_ID = e.EMPLOYEE_ID
UNION ALL
SELECT a.AO_ID AS REQUEST_ID, 'Actual Overtime' AS TYPE, e.EMPLOYEE_NAME, a.FILLING_DATE,  a.STATUS AS STATUS FROM ACTUAL_OVERTIME a
INNER JOIN PLANNED_OVERTIME p
  ON p.PO_ID = a.PO_ID
INNER JOIN EMPLOYEE e
  ON p.EMPLOYEE_ID = e.EMPLOYEE_ID
ORDER BY FILLING_DATE;

```
